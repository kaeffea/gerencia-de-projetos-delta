TRANSICOES_NDVI = {
  (SEM_LEITURA, COLETANDO):
      "dispararAtualizacaoNDVI()",
  (COLETANDO, GERANDO):
      "imagem e NDVI recebidos",
  (GERANDO, COM_RECOMENDACAO):
      "recomendacao recebida",
  (COM_RECOMENDACAO, COLETANDO):
      "[a cada X dias]",
}

def mudar_estado_ndvi(self, novo):
    # so aceita as setas que existem no DE
    if (self.estado_ndvi, novo) not in TRANSICOES_NDVI:
        raise RuntimeError("Transicao invalida")
    self.estado_ndvi = novo
